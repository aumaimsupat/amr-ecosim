ค่าพารามิเตอร์บนหน้าจอเป็นสเกลจำลอง 0–1 ไม่ได้แปลงเป็นหน่วยยาเฉพาะชนิด แบบจำลองนี้ใช้ค่าคงที่อย่างง่ายเพื่อสาธิตกลไกเท่านั้น ยังไม่ได้ fit หรือ validate กับข้อมูลเชิงทดลอง ดังนั้นควรใช้เพื่อเรียนรู้ความสัมพันธ์ภายในโมเดล ไม่ใช้คาดการณ์การรักษาหรือสถานการณ์จริง

ข้อมูลจากฐานข้อมูล

แสดงตัวอย่าง 50 แถวแรกที่อ่านจาก PostgreSQL

AB_concnutrient_concpopulationtimesurv_frac

SQL: เปรียบเทียบ 2 พารามิเตอร์

SELECT ab_conc, nutrient_conc,
       COUNT(*) AS sample_count,
       AVG(surv_frac) AS mean_surv_frac
FROM amr_evol_dynamics
GROUP BY ab_conc, nutrient_conc
ORDER BY ab_conc, nutrient_conc;

SQL: แนวโน้มตามเวลา

SELECT ab_conc, nutrient_conc, time,
       AVG(surv_frac) AS mean_surv_frac
FROM amr_evol_dynamics
GROUP BY ab_conc, nutrient_conc, time
ORDER BY ab_conc, nutrient_conc, time;

SQL: ประวัติการจำลอง

SELECT id, created_at, antibiotic, nutrient, steps,
       final_resistant_fraction
FROM amr_simulation_runs
ORDER BY created_at DESC
LIMIT 10;
