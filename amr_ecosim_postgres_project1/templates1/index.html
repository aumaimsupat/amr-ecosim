SQL: เปรียบเทียบ 2 พารามิเตอร์

SELECT ab_conc, nutrient_conc,
       COUNT(*) AS sample_count,
       AVG(surv_frac) AS mean_surv_frac
FROM amr_evol_dynamics
GROUP BY ab_conc, nutrient_conc
ORDER BY ab_conc, nutrient_conc;

SQL: แนวโน้มตามเวลา

SELECT ab_conc, nutrient_conc, time,
       AVG(surv_frac) AS mean_surv_frac
FROM amr_evol_dynamics
GROUP BY ab_conc, nutrient_conc, time
ORDER BY ab_conc, nutrient_conc, time;
